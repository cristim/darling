# LOCAL OBSERVATION PROBE, never published.
project(probe)
include(darling_lib)
set(DYLIB_COMPAT_VERSION "1.0.0")
set(DYLIB_CURRENT_VERSION "1.0.0")
set(FRAMEWORK_VERSION "A")

function(probe_dylib name def path)
	set(DYLIB_INSTALL_NAME "${path}")
	add_darling_library(${name} SHARED private.m)
	target_compile_definitions(${name} PRIVATE ${def})
	target_compile_options(${name} PRIVATE -fobjc-arc -Wno-incompatible-pointer-types)
	make_fat(${name})
	target_link_libraries(${name} system objc CoreFoundation Foundation)
	set_target_properties(${name} PROPERTIES OUTPUT_NAME ${name})
endfunction()

probe_dylib(LaunchServicesSupport PROBE_LSS /usr/lib/libLaunchServicesSupport.dylib)
probe_dylib(systemstats PROBE_SYSSTATS /usr/lib/libsystemstats.dylib)
foreach(fw RemoteServiceDiscovery:PROBE_RSD SafeEjectGPU:PROBE_SAFEEJECT AssetCacheServicesExtensions:PROBE_ACSE AOSKit:PROBE_AOSKIT Suggestions:PROBE_SUGGESTIONS)
	string(REPLACE ":" ";" parts ${fw})
	list(GET parts 0 n)
	list(GET parts 1 d)
	probe_dylib(${n} ${d} /System/Library/PrivateFrameworks/${n}.framework/Versions/A/${n})
endforeach()
